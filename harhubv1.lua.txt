local function corner(p, r)
    local c = Instance.new("UICorner", p)
    c.CornerRadius = UDim.new(0, r or 8)
    return c
end

local function stroke(p, col, th)
    local s = Instance.new("UIStroke", p)
    s.Color = col or T.STROKE ; s.Thickness = th or 1
    return s
end

local function tw(obj, props, t, sty, dir)
    TweenService:Create(obj, TweenInfo.new(t or 0.18, sty or Enum.EasingStyle.Quad, dir or Enum.EasingDirection.Out), props):Play()
end

local function makeDraggable(frame, dragZone)
    local dragging, dragStart, startPos = false, nil, nil
    dragZone.InputBegan:Connect(function(inp)
        if (inp.UserInputType == Enum.UserInputType.MouseButton1 or inp.UserInputType == Enum.UserInputType.Touch) then
            dragging = true ; dragStart = inp.Position ; startPos = frame.Position
            inp.Changed:Connect(function() if inp.UserInputState == Enum.UserInputState.End then dragging = false end end)
        end
    end)
    UserInputService.InputChanged:Connect(function(inp)
        if dragging and (inp.UserInputType == Enum.UserInputType.MouseMovement or inp.UserInputType == Enum.UserInputType.Touch) then
            local d = inp.Position - dragStart
            frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end
    end)
end

-- [ RESET HITBOX ]
local function resetHitbox()
    for part, data in pairs(_G.OriginalParts) do
        if part and part.Parent then
            part.Size = data.Size ; part.Transparency = data.Transparency
            part.Color = data.Color ; part.CanCollide = data.CanCollide
        end
    end
    _G.OriginalParts = {}
end

-- [ ESP NGƯỜI CHƠI ]
local function removeDot(p)
    if p.Character and p.Character:FindFirstChild("Head") then
        local h = p.Character.Head:FindFirstChild("DotESP")
        if h then h:Destroy() end
    end
end

local function createDot(p)
    -- Nếu tắt ESP hoặc là chính mình thì không tạo
    if not _G.ESPEnabled or p == LocalPlayer then return end
    
    -- Đợi nhân vật tải xong
    if not p.Character or not p.Character:FindFirstChild("Head") then return end
    
    removeDot(p) -- Xóa cái cũ nếu có
    
    local head = p.Character.Head
    local bill = Instance.new("BillboardGui", head)
    bill.Name = "DotESP"
    bill.Size = UDim2.new(0, 160, 0, 50)
    bill.StudsOffset = Vector3.new(0, 3, 0)
    bill.AlwaysOnTop = true
    bill.LightInfluence = 0 -- Giúp chữ luôn sáng rõ

    local lbl = Instance.new("TextLabel", bill)
    lbl.Size = UDim2.new(1, 0, 1, 0)
    lbl.BackgroundTransparency = 1 -- XÓA NỀN ĐEN
    lbl.TextScaled = true
    lbl.Font = Enum.Font.GothamBold
    lbl.TextStrokeTransparency = 0 -- Giữ viền chữ để dễ nhìn trên mọi địa hình
    lbl.TextStrokeColor3 = Color3.new(0, 0, 0)
    
    -- Màu sắc theo team
    lbl.TextColor3 = (p.Team and p.Team.Name:lower():find("marine")) 
        and Color3.fromRGB(60, 180, 255) or T.RED2

    -- Vòng lặp cập nhật khoảng cách và máu
    task.spawn(function()
        while bill.Parent and _G.ESPEnabled and task.wait(0.1) do
            local char = p.Character
            local hum = char and char:FindFirstChildOfClass("Humanoid")
            local root = char and char:FindFirstChild("HumanoidRootPart")
            local myR = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
            
            if hum and root and myR then
                local dist = math.floor((myR.Position - root.Position).Magnitude)
                lbl.Text = ("%s\n[%dm] HP:%d"):format(p.Name, dist, math.floor(hum.Health))
            else
                lbl.Text = p.Name
            end
        end
        bill:Destroy()
    end)
end

local FruitESPTable = {}

-- Logic lọc độ hiếm từ Script #2 (Dùng bảng để quản lý sạch hơn)
local MythicalFruits = {"gravity", "mammoth", "shadow", "t-rex", "mochi", "venom", "gas", "control", "tiger", "kitsune", "dragon"}
local LegendaryFruits = {"quake", "buddha", "love", "spider", "phoenix", "portal", "rumble", "pain", "blizzard"}

local function getFruitColor(name)
    name = name:lower()
    for _, v in pairs(MythicalFruits) do
        if name:find(v) then return Color3.fromRGB(255, 0, 0) end
    end
    for _, v in pairs(LegendaryFruits) do
        if name:find(v) then return Color3.fromRGB(255, 0, 255) end
    end
    return Color3.new(1, 1, 1)
end

-- Logic kiểm tra vật phẩm là Trái ác quỷ (Script #2)
local function isFruit(obj)
    return obj:IsA("Tool") and obj:FindFirstChild("Handle") and not obj:FindFirstChildOfClass("Humanoid")
end

-- Logic tạo ESP khớp với UI của Script #1
local function createFruitESP(obj)
    if not isFruit(obj) or FruitESPTable[obj] then return end

    local bill = Instance.new("BillboardGui", game.CoreGui)
    bill.Name = "FruitESP_Label"
    bill.Size = UDim2.new(0, 200, 0, 50)
    bill.Adornee = obj.Handle
    bill.AlwaysOnTop = true

    local txt = Instance.new("TextLabel", bill)
    txt.Size = UDim2.new(1, 0, 1, 0)
    txt.BackgroundTransparency = 1
    txt.TextStrokeTransparency = 0 -- Giúp chữ dễ đọc hơn (từ script #2)
    txt.TextScaled = true
    txt.Font = Enum.Font.GothamBold
    txt.Text = obj.Name
    txt.TextColor3 = getFruitColor(obj.Name)

    FruitESPTable[obj] = {Bill = bill, Text = txt}
end

for _, v in pairs(workspace:GetDescendants()) do
    if _G.FruitESPEnabled then
        createFruitESP(v)
    end
end

workspace.DescendantAdded:Connect(function(v)
    task.wait(0.1)

    if _G.FruitESPEnabled then
        createFruitESP(v)
    end
end)
-- xoa esp khi tat
local function clearFruitESP()
    for obj, data in pairs(FruitESPTable) do
        if data.Bill then
            data.Bill:Destroy()
        end
    end

    FruitESPTable = {}
end

-- Logic dọn dẹp khi trái ác quỷ bị nhặt mất
workspace.DescendantRemoving:Connect(function(v)
    if FruitESPTable[v] then
        FruitESPTable[v].Bill:Destroy()
        FruitESPTable[v] = nil
    end
end)

local function getNearestFruit()
    local char = LocalPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")

    if not root then
        return nil
    end

    local nearest = nil
    local distance = math.huge

    for _, v in pairs(workspace:GetDescendants()) do
        if isFruit(v) and v:FindFirstChild("Handle") then
            local dist = (root.Position - v.Handle.Position).Magnitude

            if dist < distance then
                distance = dist
                nearest = v
            end
        end
    end

    return nearest
end

local function createToggle(parent, text, callback)
    local row = Instance.new("Frame", parent) ; row.Size=UDim2.new(0.95,0,0,42) ; row.BackgroundColor3=T.PANEL ; corner(row,8) ; stroke(row,T.STROKE,1)

    local tbg = Instance.new("Frame",row) ; tbg.Size=UDim2.new(0,44,0,24) ; tbg.Position=UDim2.new(1,-54,0.5,-12) ; tbg.BackgroundColor3=T.GRAY ; corner(tbg,12)
    local knob = Instance.new("Frame",tbg) ; knob.Size=UDim2.new(0,18,0,18) ; knob.Position=UDim2.new(0,3,0.5,-9) ; knob.BackgroundColor3=T.WHITE ; corner(knob,9)
    
    local state = false
    local startPos = nil -- Lưu vị trí bắt đầu chạm

    row.InputBegan:Connect(function(inp)
        if inp.UserInputType == Enum.UserInputType.MouseButton1 or inp.UserInputType == Enum.UserInputType.Touch then
            startPos = inp.Position
        end
    end)

    row.InputEnded:Connect(function(inp)
        if (inp.UserInputType == Enum.UserInputType.MouseButton1 or inp.UserInputType == Enum.UserInputType.Touch) and startPos then
            local diff = (inp.Position - startPos).Magnitude
            if diff < 5 then -- Nếu di chuyển ít (Click), thì mới thực thi
                state = not state
                tw(tbg, {BackgroundColor3 = state and T.GREEN or T.GRAY}, 0.15)
                tw(knob, {Position = state and UDim2.new(1,-21,0.5,-9) or UDim2.new(0,3,0.5,-9)}, 0.15)
                callback(state)
            end
            startPos = nil
        end
    end)
end

local function createSlider(parent, text, min, max, default, callback)
    local row = Instance.new("Frame", parent) ; row.Size=UDim2.new(0.95,0,0,50) ; row.BackgroundColor3=T.PANEL ; corner(row,8) ; stroke(row,T.STROKE,1)
    local lbl = Instance.new("TextLabel",row) ; lbl.Size=UDim2.new(1,0,0,20) ; lbl.Position=UDim2.new(0,12,0,5) ; lbl.BackgroundTransparency=1 ; lbl.Text=text..": "..default ; lbl.TextColor3=T.WHITE ; lbl.TextXAlignment=Enum.TextXAlignment.Left
    local bar = Instance.new("Frame",row) ; bar.Size=UDim2.new(0.9,0,0,4) ; bar.Position=UDim2.new(0.05,0,0.75,0) ; bar.BackgroundColor3=T.GRAY ; corner(bar,2)
    local fill = Instance.new("Frame",bar) ; fill.Size=UDim2.new((default-min)/(max-min),0,1,0) ; fill.BackgroundColor3=T.RED ; corner(fill,2)
    
    local function update(input)
        local per = math.clamp((input.Position.X - bar.AbsolutePosition.X) / bar.AbsoluteSize.X, 0, 1)
        fill.Size = UDim2.new(per, 0, 1, 0)
        local val = math.floor(min + (max - min) * per)
        lbl.Text = text..": "..val ; callback(val)
    end

    local dragging = false
    local startPos = nil

    row.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            startPos = input.Position
            -- Delay nhẹ để kiểm tra xem là đang giữ để kéo hay chỉ là chạm qua để cuộn
            task.delay(0.1, function() if startPos then dragging = true end end)
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            update(input)
        end
    end)

    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            if not dragging and startPos then
                local diff = (input.Position - startPos).Magnitude
                if diff < 5 then update(input) end -- Click vào thanh bar vẫn nhận giá trị
            end
            dragging = false ; startPos = nil
        end
    end)
end

    local bar = Instance.new("Frame",row) ; bar.Size=UDim2.new(0.9,0,0,4) ; bar.Position=UDim2.new(0.05,0,0.75,0) ; bar.BackgroundColor3=T.GRAY ; corner(bar,2)
    local fill = Instance.new("Frame",bar) ; fill.Size=UDim2.new((default-min)/(max-min),0,1,0) ; fill.BackgroundColor3=T.RED ; corner(fill,2)
    
    local function update(input)
        local per = math.clamp((input.Position.X - bar.AbsolutePosition.X) / bar.AbsoluteSize.X, 0, 1)
        fill.Size = UDim2.new(per, 0, 1, 0)
        local val = math.floor(min + (max - min) * per)
        lbl.Text = text..": "..val ; callback(val)
    end
    local dragging = false
    row.InputBegan:Connect(function(input) if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then dragging = true update(input) end end)
    UserInputService.InputChanged:Connect(function(input) if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then update(input) end end)
    UserInputService.InputEnded:Connect(function(input) if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then dragging = false end end)
end

RunService.RenderStepped:Connect(function()
    local char = LocalPlayer.Character ; local hum = char and char:FindFirstChild("Humanoid") ; local root = char and char:FindFirstChild("HumanoidRootPart")
    if hum and root then
        hum.WalkSpeed = _G.SpeedEnabled and _G.SpeedValue or 50
        if _G.JumpEnabled then
            hum.UseJumpPower = true
            hum.JumpPower = _G.JumpPowerValue
        else
            hum.JumpPower = 50 -- Giá trị mặc định của Roblox
        end
        if _G.DashEnabled and hum.MoveDirection.Magnitude > 0 then root.CFrame = root.CFrame + hum.MoveDirection * _G.DashDist end
        if _G.AutoTeleEnabled then
            for _, p in pairs(Players:GetPlayers()) do
                if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
                    root.CFrame = p.Character.HumanoidRootPart.CFrame * CFrame.new(0,0,3) ; break
                end
            end
        end
    end
    if _G.HitboxEnabled then
        for _, p in pairs(Players:GetPlayers()) do
            if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
                local part = p.Character.HumanoidRootPart
                if not _G.OriginalParts[part] then _G.OriginalParts[part] = {Size = part.Size, Transparency = part.Transparency, Color = part.Color, CanCollide = part.CanCollide} end
                part.Size = Vector3.new(_G.HitboxSize, _G.HitboxSize, _G.HitboxSize) ; part.Transparency = 0.7 ; part.CanCollide = false
            end
        end
    end
    -- Fruit ESP Loop
    if _G.FruitESPEnabled then
        for _, v in pairs(workspace:GetDescendants()) do
            if v:IsA("Tool") and v:FindFirstChild("Handle") and not v:FindFirstChildOfClass("Humanoid") then createFruitESP(v) end
        end
    end
end)

UserInputService.JumpRequest:Connect(function()
    if _G.InfiniteJumpEnabled and LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Humanoid") then
        LocalPlayer.Character.Humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
    end
end)

-- Fruit Tween
if _G.FruitTweenEnabled then
    local fruit = getNearestFruit()

    if fruit and fruit:FindFirstChild("Handle") then
        local targetPos = fruit.Handle.Position + Vector3.new(0,3,0)

        root.CFrame = root.CFrame:Lerp(
            CFrame.new(targetPos),
            _G.FruitTweenSpeed / 10000
        )
    end
end
